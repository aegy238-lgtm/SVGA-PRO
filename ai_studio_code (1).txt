import React, { useState, useRef } from 'react';
import { 
  Upload, Layers, Download, Play, Pause, Eye, EyeOff, Lock, Unlock, 
  Package, FolderPlus, Video, AlertCircle, RefreshCw, X, Sparkles 
} from 'lucide-react';
import { Parser } from 'svga.lite';

// --- 1. Types ---
export interface KeyframeData {
  alpha?: number;
  transform?: { a: number; b: number; c: number; d: number; tx: number; ty: number; };
  layout?: { x: number; y: number; width: number; height: number; };
  shapes?: any[];
  clipPath?: string;
}

export interface EditableLayer {
  id: string;
  name: string;
  type: 'image' | 'shape' | 'video' | 'text';
  visible: boolean;
  locked: boolean;
  opacity: number;
  imageKey: string;
  thumbnailUrl?: string;
  width?: number;
  height?: number;
  inFrame?: number;
  outFrame?: number;
  framesCount?: number;
  transform?: { x: number; y: number; scaleX: number; scaleY: number; rotation: number; };
  spriteRef?: { imageKey?: string; frames?: KeyframeData[]; };
}

export interface SVGAProjectData {
  id: string;
  name: string;
  fileSize?: number;
  version: string;
  videoSize: { width: number; height: number; };
  fps: number;
  frames: number;
  images: { [key: string]: string };
  sprites: any[];
}

// --- 2. Engines ---
const svgaParser = new Parser();

export async function parseSvgaToProject(file: File): Promise<{ project: SVGAProjectData; layers: EditableLayer[]; }> {
  const arrayBuffer = await file.arrayBuffer();
  const svgaData: any = await svgaParser.load(arrayBuffer);

  const imagesMap: { [key: string]: string } = {};
  if (svgaData.images) {
    for (const [key, val] of Object.entries(svgaData.images)) {
      imagesMap[key] = typeof val === 'string' && val.startsWith('data:') ? val : `data:image/png;base64,${btoa(String.fromCharCode(...new Uint8Array(val as any)))}`;
    }
  }

  const project: SVGAProjectData = {
    id: `p_${Date.now()}`,
    name: file.name,
    version: svgaData.version || '2.0',
    videoSize: { width: svgaData.videoSize?.width || 750, height: svgaData.videoSize?.height || 1334 },
    fps: svgaData.fps || 30,
    frames: svgaData.frames || 60,
    images: imagesMap,
    sprites: svgaData.sprites || []
  };

  const layers: EditableLayer[] = (svgaData.sprites || []).map((sprite: any, idx: number) => ({
    id: `l_${idx}_${Date.now()}`,
    name: sprite.imageKey || `layer_${idx}`,
    type: sprite.imageKey ? 'image' : 'shape',
    visible: true,
    locked: false,
    opacity: 1,
    imageKey: sprite.imageKey || '',
    thumbnailUrl: imagesMap[sprite.imageKey],
    spriteRef: { imageKey: sprite.imageKey, frames: sprite.frames || [] }
  }));

  return { project, layers };
}

// --- 3. UI Component ---
export const SvgaLayerEditor: React.FC = () => {
  const [currentProject, setCurrentProject] = useState<SVGAProjectData | null>(null);
  const [layers, setLayers] = useState<EditableLayer[]>([]);
  const [isLoading, setIsLoading] = useState(false);
  const [isPlaying, setIsPlaying] = useState(false);
  const fileInputRef = useRef<HTMLInputElement>(null);

  const handleLoadFiles = async (files: FileList | null) => {
    if (!files || files.length === 0) return;
    setIsLoading(true);
    try {
      const { project, layers } = await parseSvgaToProject(files[0]);
      setCurrentProject(project);
      setLayers(layers);
    } catch (e) {
      alert('خطأ في تحميل الملف');
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div className="w-full h-screen bg-[#070b14] text-white flex flex-col font-sans" dir="rtl">
      <header className="px-6 py-4 border-b border-white/10 bg-slate-900/60 flex items-center justify-between">
        <h1 className="text-lg font-black">محرر وفك طبقات SVGA الاحترافي</h1>
        {currentProject && <button onClick={() => setCurrentProject(null)} className="bg-indigo-600 px-4 py-2 rounded-xl text-xs font-bold cursor-pointer">جديد</button>}
      </header>

      <div className="flex-1 flex overflow-hidden">
        {currentProject ? (
          <div className="flex-1 flex">
            <div className="flex-1 flex flex-col items-center justify-center bg-black">
              <div className="w-[300px] h-[500px] bg-slate-900 flex items-center justify-center border border-white/10 rounded-xl relative">
                 {layers.filter(l => l.visible).map(l => l.thumbnailUrl && <img key={l.id} src={l.thumbnailUrl} className="absolute inset-0 w-full h-full object-contain" />)}
              </div>
              <button onClick={() => setIsPlaying(!isPlaying)} className="mt-4 p-4 bg-indigo-600 rounded-full cursor-pointer">{isPlaying ? <Pause /> : <Play />}</button>
            </div>
            <div className="w-80 bg-slate-900 p-4 overflow-y-auto">
              {layers.map(l => (
                <div key={l.id} className="p-2 mb-2 bg-black/50 rounded flex justify-between items-center text-xs">
                  <span>{l.name}</span>
                  <div className="flex gap-2">
                    <button onClick={() => setLayers(prev => prev.map(ly => ly.id === l.id ? {...ly, visible: !ly.visible} : ly))}>{l.visible ? <Eye size={14}/> : <EyeOff size={14}/>}</button>
                  </div>
                </div>
              ))}
            </div>
          </div>
        ) : (
          <div className="flex-1 flex flex-col items-center justify-center">
            <input type="file" ref={fileInputRef} accept=".svga" className="hidden" onChange={(e) => handleLoadFiles(e.target.files)} />
            <button onClick={() => fileInputRef.current?.click()} className="p-10 bg-indigo-600/20 border border-indigo-500 rounded-3xl flex flex-col items-center gap-4 cursor-pointer">
              <Upload size={40} />
              <span className="font-bold">اضغط هنا لرفع ملف SVGA</span>
            </button>
          </div>
        )}
      </div>
    </div>
  );
};
